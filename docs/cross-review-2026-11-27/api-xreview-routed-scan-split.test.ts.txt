import { DatabaseId, DEFAULT_DATABASE_OPTIONS, FULL_KEY_RANGE } from "@shaledb/lsm-core";
import {
  DatabaseLifecycle,
  DbClient,
  dbServicesTestLayer,
  lsmTestRuntimeContextLayer,
  makeDbClient,
  ScanPageBudget,
  WriterClient,
} from "@shaledb/lsm-cloudflare";
import { Effect, Layer, Option, Stream } from "effect";
import { describe, expect, it } from "vite-plus/test";

import { ShardDirectory, shardDirectoryTestLayer } from "./shard-directory.ts";
import { ShardGroup } from "./shard-map.ts";
import { makeShardRouter, shardRouterLayerWithoutDependencies } from "./shard-router.ts";
import { ShardSplit, shardSplitLayerWithoutDependencies } from "./shard-split.ts";

/** The sharded scan's work per page (D125): limit 1 must not read a shard's whole range. */
const GROUP = ShardGroup.make("orders");
const FIRST = DatabaseId.make("orders-0");
const key = (index: number) => Uint8Array.of(0x6b, index >> 8, index & 0xff);

const testLayer = shardDirectoryTestLayer.pipe(
  Layer.provideMerge(dbServicesTestLayer({ blockSize: 4096 })),
);

const splitLayer = Layer.mergeAll(
  shardRouterLayerWithoutDependencies,
  shardSplitLayerWithoutDependencies,
).pipe(
  Layer.provideMerge(shardDirectoryTestLayer),
  Layer.provideMerge(dbServicesTestLayer({ blockSize: 4096 })),
);


/**
 * Cross-review (xreview-api): since D93 a Latest scan reads the writer's pages lazily, so a shard
 * sealed by a split during the scan reports it (DatabaseSealed) at a later page, after rows were
 * returned. The router's routing-error retry (D57, D59) then scans its whole range again on the
 * new map: the rows already returned come back a second time. Before D93 every writer page was
 * read before the first row, so the refusal always came first.
 */
describe("xreview-api: a routed Latest scan that meets a split between writer pages", () => {
  it("returns each key once, in order", async () => {
    const keys = await Effect.gen(function* () {
      yield* (yield* DatabaseLifecycle).create(FIRST, {
        keyRange: Option.none(),
        databaseOptions: DEFAULT_DATABASE_OPTIONS,
      });
      yield* (yield* ShardDirectory).create(GROUP, FIRST);
      const db = yield* DbClient;
      // 2,500 live keys, unflushed: three writer pages.
      for (let start = 0; start < 2500; start += 500) {
        yield* db.write(FIRST, {
          operations: Array.from({ length: 500 }, (_, offset) => ({
            _tag: "Put" as const,
            key: key(start + offset),
            value: Uint8Array.of(1),
            ttlMillis: Option.none(),
          })),
        });
      }
      const writer = yield* WriterClient;
      const splitter = yield* ShardSplit;
      let calls = 0;
      // The split lands between the scan's first and second writer pages.
      const client = yield* makeDbClient.pipe(
        Effect.provideService(
          WriterClient,
          WriterClient.of({
            ...writer,
            scan: (databaseId, request) =>
              Effect.gen(function* () {
                calls += 1;
                if (calls === 2 && databaseId === FIRST) {
                  yield* splitter.split(GROUP, FIRST, key(2000));
                }
                return yield* writer.scan(databaseId, request);
              }),
          }),
        ),
      );
      const router = yield* makeShardRouter.pipe(Effect.provideService(DbClient, client));
      const rows = yield* Stream.runCollect(
        router.scan(GROUP, FULL_KEY_RANGE, {
          consistency: { _tag: "Latest" },
          snapshot: Option.none(),
          refuseSealed: true,
          limit: Option.none(),
        }),
      );
      return { indexes: rows.map((row) => ((row.key[1] ?? 0) << 8) | (row.key[2] ?? 0)), calls };
    }).pipe(
      Effect.scoped,
      Effect.provide(splitLayer),
      Effect.provide(lsmTestRuntimeContextLayer),
      Effect.runPromise,
    );
    expect(keys.calls).toBeGreaterThanOrEqual(2);
    expect(keys.indexes.length).toBe(2500);
    expect(keys.indexes).toEqual(Array.from({ length: 2500 }, (_, index) => index));
  }, 120_000);
});
